use barbearia;

CREATE TABLE agendamentos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cliente VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    servico VARCHAR(100),
    preco DECIMAL(10 , 2 ),
    barbeiro VARCHAR(50),
    data_agendamento DATE NOT NULL,
    horario VARCHAR(5),
    status_agendamento ENUM('Agendado', 'Concluído', 'Cancelado') DEFAULT 'Agendado'
);
